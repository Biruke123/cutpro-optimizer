function drawCuttingDiagram(data) { var container = document.getElementById('resultsContent'); var existingCanvas = document.getElementById('cuttingCanvas'); if (existingCanvas) existingCanvas.remove(); var details = data.details; var allSheets = []; Object.keys(details).forEach(function(materialKey) { var materialData = details[materialKey]; if (materialData && Array.isArray(materialData.sheets)) { materialData.sheets.forEach(function(sheet) { if (sheet && !sheet.is_scrap_note) { allSheets.push(sheet); } }); } }); if (!allSheets || allSheets.length === 0) { var warningCanvas = document.createElement('canvas'); warningCanvas.id = 'cuttingCanvas'; warningCanvas.width = 800; warningCanvas.height = 200; warningCanvas.style.width = '100%'; warningCanvas.style.border = '1px solid #d0d0d0'; warningCanvas.style.background = '#ffffff'; container.appendChild(warningCanvas); var warnCtx = warningCanvas.getContext('2d'); warnCtx.fillStyle = '#ff0000'; warnCtx.font = 'bold 20px Arial'; warnCtx.textAlign = 'center'; warnCtx.textBaseline = 'middle'; warnCtx.fillText("⚠️ No sheets found.", 400, 100); return; } var canvas = document.createElement('canvas'); canvas.id = 'cuttingCanvas'; canvas.width = 900; var canvasHeight = (allSheets.length * 350) + 50; canvas.height = canvasHeight; canvas.style.width = '100%'; canvas.style.height = 'auto'; canvas.style.borderRadius = '6px'; canvas.style.border = '1px solid #d0d0d0'; canvas.style.marginTop = '16px'; canvas.style.background = '#ffffff'; container.appendChild(canvas); var ctx = canvas.getContext('2d'); var sheetWidth = data.sheet_width || 2440; var sheetHeight = data.sheet_height || 1220; var scaleX = (canvas.width - 100) / sheetWidth; var scaleY = 170 / sheetHeight; var scale = Math.min(scaleX, scaleY, 0.35); var yOffset = 60; var partColors = [ '#FDEFF2', '#F0F8FF', '#F5F0FA', '#F0FBF0', '#FFF5EC', '#EEF6FB', '#FDF3F2', '#EAF5F8', '#FBF7EC', '#F5EEF7', '#FFEEEE', '#EEFFEE', '#EEEEFF', '#FFFFEE', '#FFEEFF', '#EEFFFF' ]; ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, canvas.width, canvas.height); allSheets.forEach(function(sheet, idx) { var sWidth = parseFloat(sheet.sheet_width) || 2440; var sHeight = parseFloat(sheet.sheet_height) || 1220; var x = 50; var y = yOffset; var w = sWidth * scale; var h = sHeight * scale; ctx.fillStyle = '#ffffff'; ctx.fillRect(x, y, w, h); ctx.strokeStyle = '#000000'; ctx.lineWidth = 2.5; ctx.strokeRect(x, y, w, h); ctx.fillStyle = '#000000'; ctx.font = 'bold 14px Arial'; ctx.textAlign = 'left'; ctx.textBaseline = 'bottom'; ctx.fillText('Sheet ' + (sheet.sheet_number || (idx + 1)), x, y - 10); ctx.fillStyle = '#333333'; ctx.font = '12px Arial'; ctx.textAlign = 'center'; ctx.textBaseline = 'bottom'; ctx.fillText(Math.round(sWidth) + ' × ' + Math.round(sHeight) + ' mm', x + w/2, y - 10); var sheetParts = sheet.parts || []; var sortedParts = sheetParts.sort(function(a, b) { return (a.y || 0) - (b.y || 0) || (a.x || 0) - (b.x || 0); }); sortedParts.forEach(function(part, pIdx) { var partX = parseFloat(part.x) || 0; var partY = parseFloat(part.y) || 0; var partW = parseFloat(part.w) || 0; var partH = parseFloat(part.h) || 0; var px = x + partX * scale; var py = y + partY * scale; var pw = partW * scale; var ph = partH * scale; if (pw < 2 || ph < 2) return; // --- SMART LABEL --- var smartLabel = part.label || 'PART'; var ratio = Math.max(partW, partH) / Math.min(partW, partH); if (partW > 500 && partH > 500 && ratio > 1.2) smartLabel = 'DOOR'; else if (ratio > 2.0 && partW > 200) smartLabel = 'SIDE PANEL'; else if (partW < 700 && partH < 700) smartLabel = 'DRAWER'; else smartLabel = 'SHELF'; // ========================================================= // 🔥 FIX: ALL DOOR AND DRAWER PARTS GET VENEER HIGHLIGHT 🔥 // ========================================================= var isHighlighted = false; // Check if this part is a DOOR or DRAWER (they always get veneer) var label = (part.label || smartLabel || '').toUpperCase(); if (label.indexOf('DOOR') !== -1 || label.indexOf('DRAWER') !== -1) { isHighlighted = true; } // ALSO check if the material is actually a veneer if (!isHighlighted && typeof veneerHighlights !== 'undefined' && veneerHighlights.length > 0) { for (var i = 0; i < veneerHighlights.length; i++) { if (veneerHighlights[i].material === part.material) { isHighlighted = true; break; } } } // 1. Draw the colored background ctx.fillStyle = partColors[pIdx % partColors.length]; ctx.fillRect(px, py, pw, ph); // 2. Draw the outer solid border ctx.strokeStyle = '#000000'; ctx.lineWidth = 1.0; ctx.setLineDash([]); ctx.strokeRect(px, py, pw, ph); // 3. DRAW THE DASHED MARKER (Only for Veneered parts) if (isHighlighted) { ctx.strokeStyle = '#000000'; ctx.lineWidth = 1.5; ctx.setLineDash([4, 4]); ctx.strokeRect(px + 3, py + 3, pw - 6, ph - 6); ctx.setLineDash([]); } // 4. Draw the text if (pw > 40 && ph > 20) { var displayText = smartLabel + ' ' + Math.round(partW) + '×' + Math.round(partH); ctx.fillStyle = '#000000'; ctx.font = 'bold 10px Arial'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; if (ctx.measureText(displayText).width > pw - 10) { ctx.font = 'bold 8px Arial'; } ctx.fillText(displayText, px + pw/2, py + ph/2); } }); yOffset += h + 90; }); }
